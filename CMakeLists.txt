# Host builds exercise portable firmware logic; ESP-IDF builds use the same sources.
cmake_minimum_required(VERSION 3.20)

if(DEFINED ENV{IDF_PATH} AND (DEFINED ESP_PLATFORM OR CMAKE_TOOLCHAIN_FILE MATCHES "esp-idf"))
    include($ENV{IDF_PATH}/tools/cmake/project.cmake)
    project(mainboard_firmware VERSION 1)
    set(Z1_WEBUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/webui" CACHE PATH
        "Directory whose contents are packed into the SPIFFS partition")
    if(NOT IS_DIRECTORY "${Z1_WEBUI_DIR}")
        message(FATAL_ERROR "Web UI source directory does not exist: ${Z1_WEBUI_DIR}")
    endif()
    spiffs_create_partition_image(spiffs "${Z1_WEBUI_DIR}" FLASH_IN_PROJECT)
else()
    project(mainboard_firmware_host LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    enable_testing()
    add_subdirectory(components/core)
    add_subdirectory(components/application)
    add_subdirectory(tests)
endif()
